Common Route Patterns

// Basic route
Route::get('/products', [ProductController::class, 'index']);

// API resource (automatic CRUD routes)
Route::apiResource('/products', ProductController::class);

// Nested resource
Route::apiResource('/users.tasks', TaskController::class);

// Versioned API
Route::prefix('v1')->group(function () {
    Route::apiResource('products', ProductController::class);
});

// With middleware
Route::middleware('auth:sanctum')->group(function () {
    Route::apiResource('products', ProductController::class);
});

Eager Loading (Avoid N+1)

// Bad — N+1 problem
$users = User::all();
foreach ($users as $user) { echo $user->posts; }

// Good — Eager loading
$users = User::with('posts')->get();

// Multiple relationships
$users = User::with(['posts', 'profile'])->get();

// Nested eager loading
$users = User::with('posts.comments')->get();

5. Middleware Creation Steps

// Step 1: Create the middleware
php artisan make:middleware CheckRole

// Step 2: Define handle() logic
// app/Http/Middleware/CheckRole.php
public function handle(Request $request, Closure $next, string $role)
{
    if (!$request->user() || !$request->user()->hasRole($role)) {
        return response()->json(['message' => 'Unauthorized.'], 403);
    }
    return $next($request);
}

// Step 3: Register in Kernel.php
// app/Http/Kernel.php
protected $middlewareAliases = [
    'role' => \App\Http\Middleware\CheckRole::class,
];

// Step 4: Use in routes
Route::middleware('role:admin')->group(function () {
    Route::apiResource('products', ProductController::class);
});

6. Queue / Job Workflow

// 1. Create job
php artisan make:job ProcessPayment

// 2. Define handle() logic
public function handle(): void
{
    // Process payment logic here
}

// 3. Dispatch the job
ProcessPayment::dispatch($order);

// With delay
ProcessPayment::dispatch($order)->delay(now()->addMinutes(10));

// On specific queue
ProcessPayment::dispatch($order)->onQueue('payments');

// 4. Run queue worker
php artisan queue:work

// 5. Failed jobs
php artisan queue:failed         // List failed
php artisan queue:retry all      // Retry all
php artisan queue:flush          // Clear all

// 6. Table for failed jobs
php artisan queue:failed-table
php artisan migrate

Cache Usage Examples

// Store forever
Cache::forever('key', $value);

// Remember (cache or callback)
$products = Cache::remember('products.all', 3600, function () {
    return Product::all();
});

// Remember forever
Cache::rememberForever('key', fn() => expensiveQuery());

// Pull (get and delete)
$value = Cache::pull('key');

// Increment/Decrement
Cache::increment('visits');
Cache::decrement('visits', 5);

// Tags (Redis only)
Cache::tags(['products', 'active'])->flush();
